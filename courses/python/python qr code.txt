import pyotp
from qrcode import QRCode


# Account details
username = "student@example.com"
issuer = "Ledger LMS"

# Generate secret
secret = pyotp.random_base32()

# Create TOTP
totp = pyotp.TOTP(secret)

# Create Google Authenticator URI
uri = totp.provisioning_uri(
    name=username,
    issuer_name=issuer
)

print("\n" + "=" * 50)
print("       GOOGLE AUTHENTICATOR SETUP")
print("=" * 50)

print("\nSecret Key:")
print(secret)

print("\nScan this QR code using Google Authenticator:\n")


# Generate QR
qr = QRCode(
    box_size=1,
    border=1
)

qr.add_data(uri)
qr.make(fit=True)

matrix = qr.get_matrix()

# Compact QR display
for y in range(0, len(matrix), 2):
    line = ""

    for x in range(len(matrix[0])):
        top = matrix[y][x]
        bottom = matrix[y + 1][x] if y + 1 < len(matrix) else False

        if top and bottom:
            line += "█"
        elif top:
            line += "▀"
        elif bottom:
            line += "▄"
        else:
            line += " "

    print(line)


print("\n" + "=" * 50)

# Ask for OTP
user_otp = input("Enter the 6-digit OTP: ")

# Verify OTP
if totp.verify(user_otp):
    print("\n\033[92m✓ AUTHENTICATION SUCCESS\033[0m")
else:
    print("\n\033[91m✗ AUTHENTICATION FAILED\033[0m")

print("=" * 50)
